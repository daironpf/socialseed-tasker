**`scripts/wait-for-service.sh`**
```bash
#!/usr/bin/env bash
# usage: wait-for-service.sh <url> <timeout_seconds>
set -euo pipefail
URL="$1"
TIMEOUT="${2:-60}"
i=0
while true; do
  if curl -sSf "$URL" >/dev/null 2>&1; then
    echo "Service $URL is reachable"
    exit 0
  fi
  i=$((i+1))
  if [ "$i" -ge "$TIMEOUT" ]; then
    echo "Timed out waiting for $URL" >&2
    exit 1
  fi
  sleep 1
done
```
Make the script executable (`chmod +x scripts/wait-for-service.sh`).

**`examples/mini_project/bootstrap.py`**
```python
#!/usr/bin/env python3
import json
import os
import subprocess
import sys
from pathlib import Path

# This script demonstrates end-to-end usage:
# 1. Create two issues via CLI
# 2. Attach example file paths to issue metadata
# 3. Add dependency A -> B
# 4. Run agent-context for issue B and write output to examples/output.json

ROOT = Path(__file__).parent
EXAMPLE_A = ROOT / "a.py"
EXAMPLE_B = ROOT / "b.py"
OUTPUT = ROOT.parent / "output.json"

PY = sys.executable
MODULE = "-m"
MAIN = "tasker.cli.main"

def run_cmd(args):
    proc = subprocess.run([PY, MODULE, MAIN] + args, capture_output=True, text=True)
    if proc.returncode != 0:
        print("Command failed:", args, file=sys.stderr)
        print(proc.stderr, file=sys.stderr)
        sys.exit(proc.returncode)
    return proc.stdout

def create_issue(issue_id, title, files):
    # create issue via CLI
    run_cmd(["create-issue", "--id", issue_id, "--title", title])
    # update metadata by using repository directly if CLI doesn't support metadata
    # For determinism, we will call Neo4j directly via cypher using the Neo4j adapter if available
    # Fallback: use run_cypher via tasker infrastructure if importable
    try:
        from tasker.infrastructure.neo4j_adapter import Neo4jGraphAdapter
        from tasker.infrastructure.neo4j_issue_repository import Neo4jIssueRepository
        from tasker.application.dtos import IssueDTO
        graph = Neo4jGraphAdapter(uri=os.getenv("TASKER_NEO4J_URI", "bolt://localhost:7687"),
                                 user=os.getenv("TASKER_NEO4J_USER", "neo4j"),
                                 password=os.getenv("TASKER_NEO4J_PASSWORD", "test"))
        repo = Neo4jIssueRepository(graph)
        dto = IssueDTO(id=issue_id, title=title, description="", status="open", metadata={"files": files})
        repo.save(dto)
        graph.close()
    except Exception as exc:
        print("Warning: could not update metadata via repository, continuing. Error:", exc, file=sys.stderr)

def add_dependency(from_id, to_id):
    run_cmd(["add-dependency", "--from", from_id, "--to", to_id])

def generate_context(issue_id):
    out = run_cmd(["agent-context", "--issue-id", issue_id, "--max-depth", "3"])
    # CLI prints JSON to stdout on success
    j = json.loads(out)
    return j

def main():
    # Create issues
    create_issue("example-a", "Example A", [str(EXAMPLE_A)])
    create_issue("example-b", "Example B", [str(EXAMPLE_B)])
    # Add dependency example-a -> example-b
    add_dependency("example-a", "example-b")
    # Generate context for example-b
    ctx = generate_context("example-b")
    # Write output
    with open(OUTPUT, "w", encoding="utf-8") as fh:
        json.dump(ctx, fh, indent=2)
    print("Wrote example output to", OUTPUT)

if __name__ == "__main__":
    main()
```
